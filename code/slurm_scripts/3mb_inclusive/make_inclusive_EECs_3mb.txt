#!/bin/bash
#SBATCH --job-name=inc_eec_all_batches_3mb
#SBATCH --account=commons              
#SBATCH --partition=commons            
#SBATCH --ntasks=1                
#SBATCH --cpus-per-task=1             
#SBATCH --mem-per-cpu=32G           
#SBATCH --time=06:00:00            
#SBATCH --mail-user=rj65@rice.edu 
#SBATCH --mail-type=ALL            


#SBATCH --output=logs/EEC/inclusive_%j.out       
#SBATCH --error=logs/EEC/inclusive_%j.err 



cd $SHARED_SCRATCH/$USER/wta_scratch # Move your execution context safely into scratch
source $HOME/.venv/bin/activate # activate venv

echo "analysing non-binned EEC for ALL BATCHES in 3mb inclusive dataset"
echo "Starting at : $(date)"
echo "Running on hosts: $SLURM_NODELIST"
echo "Running on $SLURM_NNODES nodes."
echo "Running on $SLURM_NPROCS processors."
echo "Current WORKDIR is : $SLURM_SUBMIT_DIR"

if [[ -d $SHARED_SCRATCH/$USER && -w $SHARED_SCRATCH/$USER ]]; then

    # target ALL BATCHES
    INPUT_FILE="$SHARED_SCRATCH/$USER/wta_scratch/data/3mb/inclusive/"
    OUTPUT_DIR="$SHARED_SCRATCH/$USER/wta_scratch/intermediate_output/EEC/3mb_inclusive/premerge/"


    # submit sbatch from the same directory as the .py file
    python -u "$SHARED_SCRATCH/$USER/wta_scratch/non_binned_inclusive_EEC_maker.py" \
    --input "$INPUT_FILE" \
    --outdir "$OUTPUT_DIR" \
    --batchnum 0
fi
